# [106. Construct Binary Tree from Inorder and Postorder Traversal](https://leetcode.com/problems/construct-binary-tree-from-inorder-and-postorder-traversal/)  
目標 -> 利用Inorder和Postorder建立一棵二元樹(unique)  
ex inorder = [9,3,15,20,7], postorder = [9,15,7,20,3]  
利用它們的特性 postorder最後一個一定是root 而inorder的root左右則是左右子樹 可以得到  
[9(左),3(root),(右)15,20,7], [9(左),(右)15,7,20,3(root)] 再往下拆 左子樹沒了 剩右子樹  
[15(左),20(root),7(右)] -> [15(左),7(右),20(root)]  
下面我是直接切vector 但其實傳範圍更好 更省資源  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:

public:
    TreeNode* buildTree(vector<int>& inorder, vector<int>& postorder) {
        if ( inorder.size() == 0 ) {
            return NULL;
        }
        TreeNode* root = new TreeNode(postorder[postorder.size()-1]);
        if ( inorder.size() == 1 ) {
            return root;
        }
        int headPos = 0;
        for ( int i = 0 ; i < inorder.size() ; i++ ) {
            if ( inorder[i] == root -> val ) {
                headPos = i;
                i = inorder.size();
            }
        }

        //切割vector 我覺得應該可以不用實際切 傳範圍參數就好
        vector<int> Linorder( inorder.begin(), inorder.begin()+headPos );
        vector<int> Lpostorder( postorder.begin(), postorder.begin()+Linorder.size());

        vector<int> Rinorder( inorder.begin()+headPos+1, inorder.end() );
        vector<int> Rpostorder( postorder.begin()+Linorder.size(), postorder.end()-1 );

        root -> left = buildTree( Linorder, Lpostorder );
        root -> right = buildTree( Rinorder, Rpostorder );
        return root;
    }
};
```
