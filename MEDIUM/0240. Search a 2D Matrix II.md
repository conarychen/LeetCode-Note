# [240. Search a 2D Matrix II](https://leetcode.com/problems/search-a-2d-matrix-ii/description/)  
目標 -> 在2D矩陣內找到target值  
因為每列都是有排序過的  
所以我先看target是否在每列頭尾區間內  
可能有多列的選擇所以不只跑一次  
這邊寫得有點亂  
反正就是每列用binary search找 沒找到就換下一列  
```cpp
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
       bool result = false, isEnd = false;
       vector<int> row;
       bool findInterval = false;
       while( !isEnd ) {
            
            for ( int i = 0 ; i < matrix.size() ; i++ ) { //先抓哪些列
                if ( target >= matrix[i][0] && target <= matrix[i][matrix[i].size()-1] ) {
                    row.push_back( i );
                    findInterval = true;

                }
            }
            if ( findInterval ) { //在該列找 這邊可以優化 用binary之類的
                
                int left, right, mid;
                bool searchEnd = false;
                for ( int j = 0 ; j < row.size() && !isEnd ; j++ ) {
                    
                    left = 0;
                    right = matrix[row[j]].size()-1;
                    searchEnd = false;
                    while ( !searchEnd && !isEnd ) {
                        
                        mid = left + ( ( right - left ) / 2 );
                        if ( target == matrix[row[j]][left] || target == matrix[row[j]][right] ) {
                            searchEnd = true;
                            isEnd = true;
                            result = true;
                        }
                        else if ( left == right ) {

                            searchEnd = true;
                            if ( j == row.size()-1 ) {
                                isEnd = true;
                                result = false;
                            }

                        }
                        else {
                            if ( target < matrix[row[j]][mid] ) {
                                right = mid-1;
                            }
                            else if ( target > matrix[row[j]][mid] ) {
                                left = mid+1;
                            }
                            else {
                                searchEnd = true;
                                isEnd = true;
                                result = true;
                            }
                        }
                    }
                }

                isEnd = true; //避免沒找到所以要設定
            }
            else { //一開始就沒找到區間代表根本沒有
                isEnd = true;
                result = false;
            }
       } 
       return result;
    }
};
```
