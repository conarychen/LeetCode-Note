# [47. Permutations II](https://leetcode.com/problems/permutations-ii/)  
與[46](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0046.%20Permutations.md)類似  
差在輸入會有重複數字  
因此一樣要去重複  
這裡用usedH來記錄那些開頭用過了  
每一層都用usedRound記錄用過的數字  
每輪用used紀錄(這裡的每輪是指每個組合內用過的位置)
當每輪用過的位置 or 用過的開頭 or 這層用過的數字 就跳過這次繼續往下做  
(詳細舉例在下面註解)  

```cpp
class Solution {
private:
    vector<vector<int>> result;
    vector<int> temp;
    unordered_set<int> usedH;
    void permutation( vector<int>& nums, vector<bool>& used ) {
        if ( temp.size() == nums.size() ) {
            result.push_back( temp );
            return;
        }
        //用set紀錄每層用過的 (左側為index) 以[1,1,2]為例
        // 2,0,1 -> usedRound = { 2, 1, 1 }
        // 2,1,0 -> 當退回2,1的時候 就會因為上面那層已經用過1 而跳過
        // 2,0,1 -> [2,1,1] 2,1,0 -> [2,1,1] 是相同的組合
        unordered_set<int> usedRound;
        for ( int i = 0 ; i < nums.size() ; i++ ) {
            if ( used[i] == true || 
                ( temp.size() == 0 && usedH.find( nums[i] ) != usedH.end() ) ||
                  usedRound.find( nums[i] ) != usedRound.end() ) continue;

            if ( temp.size() == 0 ) {
                usedH.insert( nums[i] );
            }
        
            usedRound.insert(nums[i]);
            temp.push_back( nums[i] );
            used[i] = true;
            permutation( nums, used );
            temp.pop_back();
            used[i] = false;
        }
    }
public:
    vector<vector<int>> permuteUnique(vector<int>& nums) {
        vector<bool> used( nums.size(), false );
        permutation( nums, used );
        return result;
    }
};
```
