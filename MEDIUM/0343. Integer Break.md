# [343. Integer Break](https://leetcode.com/problems/integer-break/description/)  
目標 -> 把數字拆開 拆開的這些數的和等於原本的數 要求這些拆開的數最大乘積  
一樣利用DP來找出最優解 註解內有例子說明 可能比較好理解一點  
```cpp
class Solution {
public:
    int integerBreak(int n) {
        if ( n == 2 ) return 1;
        vector<int> dp( n+1, 0 );
        dp[0] = 0;
        dp[1] = 1;
        dp[2] = 1;

        for ( int i = 3 ; i <= n ; i++ ) {
            for ( int j = 1 ; j < i-1 ; j++ ) {
                dp[i] = max( dp[i], max( (i-j)*j, dp[i-j]*j ) );
            }
        }
        //dp function --> 現在的值 or ( 拆成i,j算出來的值 or i,j繼續拆可以得到更好的值 )
        // ex:4  -> dp[4] = max( 0, max( (3-1)*1, dp[3-1]*1 ) )
        //                = max( 0, max(  2     ,   1 * 1 ) )
        //                = max( 0, 2 )
        //                = 2;
        // next i = 4 , j = 1  -> dp[4] = 3   ... until j = 2
        //                = max( 3, max( ( 4-2 )*2, dp[2]*1 )
        //                = max( 3, 4 )
        // get true ans dp[4] = 4; 
        return dp[n];

    }
};
```
