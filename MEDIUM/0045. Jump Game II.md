# [45. Jump Game II](https://leetcode.com/problems/jump-game-ii/)  
目標 -> 最少用幾步可以從index 0跳到最後一格  
每格內的數字為最多可以走幾步 -> 每次可走0~nums[i]步  
利用greedy 看nums[i]+nums[i+1]哪個組合可以走最遠 就取i為多少  
(詳細在註解內)  

```cpp
class Solution {
public:
    int jump(vector<int>& nums) {
        int maxPos = 0, temp = 0, count = 0;
        int tempPos = 0;
        // greedy 
        // 2,3,1,1,4
        // maxPos == 0 -> we can choose [1,2] steps
        // and we peek which step will take us farther then we pick it
        while ( maxPos < nums.size()-1 && nums[maxPos] != 0 ) {
            temp = maxPos;
            for ( int i = 1 ; i <= nums[maxPos] ; i++ ) {
                if ( maxPos+i >= nums.size()-1 ) {
                    tempPos = nums.size();
                }
                else if ( temp < maxPos + i + nums[maxPos+i] ) {
                    temp = maxPos + i + nums[maxPos+i];
                    tempPos = maxPos+i;
                }
            }
            maxPos = tempPos;

            count++;
        }
        return count;
    }
};
```
