# [93. Restore IP Addresses](https://leetcode.com/problems/restore-ip-addresses/description/)  
目標 -> 將input的字串切成合法的IP (0\~255.0\~255.0\~255.0\~255) 將所有可能存下來  
一樣用遞迴+回朔 用temp存可能的字串 當temp有四段且讀完整個input的時候代表這可以組成合法的IP  
我是直接取substring 但其實可以不用真切 只要傳開始跟結束的位置 來操作就好 這樣會省很多資源  

```cpp
class Solution {
private:
    vector<string> result;
    vector<string> temp;
    bool isValidIP( string s ) {
        stringstream ss(s);
        int num = -1;
        if ( ss >> num && num >= 0 && num <= 255 ) {
            cout << num << endl;
            if ( (s[0] != '0' && num != 0) || s == "0" ) {
                return true;
            }

        }
        return false;
    }
    void findIP( string s, int start ) {
        if ( start >= s.size() || temp.size() >= 4 ) {
            if ( temp.size() == 4 && start == s.size() ) {
                    string temps;
                for ( int i = 0 ; i < temp.size() ; i++ ) {
                    temps += temp[i];
                    if ( i != temp.size()-1 ) {
                        temps += ".";
                    }
                }
                result.push_back( temps );
            }
            return;
        }
        // 可以直接操作s -> 在裡面加入逗點跟刪除逗點 不用切子字串
        for ( int i = start ; i < s.size() ; i++ ) {
            string str = s.substr( start, i-start+1 );
            if ( isValidIP( str ) ) {
                temp.push_back( str );
                findIP( s, i+1 );
                temp.pop_back();
            }
            
        }
    }
public:
    vector<string> restoreIpAddresses(string s) {
        findIP( s, 0 );
        return result;
    }
};
```
