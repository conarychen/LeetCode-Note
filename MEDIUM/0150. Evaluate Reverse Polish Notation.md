# [150. Evaluate Reverse Polish Notation](https://leetcode.com/problems/evaluate-reverse-polish-notation/description/)  
目標 -> 基本上是運算傳入的後序數學式  
因為大學時的PAL project寫過類lisp語言的interpreter 所以這題沒什麼難度  
這題甚至連token都切好 不需要寫parser  
簡單來說用stack放數字  
當目前字元式算數運算子(+-*/)時 就pop stack中最後兩個數字出來計算後再push回去  
頂多就是前面判斷是不是數字時要多判斷正負即可  

```cpp
class Solution {
public:

    bool isNum( string str ) {
        bool result = true;
        bool isNeg = false;
        int i = 0;
        if ( str[0] == '-' && str.size() != 1 ) {
            isNeg = true;
            i = 1;
        }
        for ( ; i < str.size() && result ; i++ ) {
            if ( !isdigit( str[i] ) ) {
                result = false;
            }
        }
        if ( isNeg ) {
            result *= -1;
        }
        return result;
    }
    int evalRPN(vector<string>& tokens) {
        vector<int> numStack;
        int numO, numT, result;
        for ( int i = 0 ; i < tokens.size() ; i++ ) {
            if ( isNum( tokens[i] ) ) { //如果是數字就丟進stack
                numStack.push_back( stoi(tokens[i]) );
            }
            else {
                numT = numStack.back();
                numO = numStack[numStack.size()-2];

                if ( tokens[i] == "+" ) {
                    result = numO + numT;
                }
                else if ( tokens[i] == "-" ) {
                    result = numO - numT;
                }
                else if ( tokens[i] == "*" ) {
                    result = numO * numT;
                }
                else if ( tokens[i] == "/" ) {
                    result = numO / numT;
                }
                else {
                    cout << "error symbol:" << tokens[i] << endl;
                }
                numStack.pop_back();
                numStack.pop_back();
                numStack.push_back( result ); 
            }
        }
        return numStack[0];
    }
};
```
