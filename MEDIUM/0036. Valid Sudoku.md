# [36. Valid Sudoku](https://leetcode.com/problems/valid-sudoku/)  
目標 -> 驗證input的數獨題目是否合規定(但不保證有解)  
分成直橫線跟9宮格檢查  
我用set來記錄用過的數字 重複就代表有錯  
9宮格部分就用每個9宮格的中心 總共9個檢查即可  

```cpp
class Solution {
public:
    bool checkLine( vector<vector<char>>& board, int m, int n ) {
        set<char> s;
        bool result = true;
        for ( int i = 0 ; i < 9 && result ; i++ ) {
            if ( board[m][i] != '.' ) {
                if ( s.find(board[m][i]) == s.end() ) {
                    s.insert( board[m][i] );
                }
                else {
                    result = false;
                }
            }      
        }
        if ( result ) {
            s.clear();
            for ( int i = 0 ; i < 9 && result ; i++ ) {
                if ( board[i][n] != '.' ) {
                    if ( s.find(board[i][n]) == s.end() ) {
                        s.insert( board[i][n] );
                    }
                    else {
                        result = false;
                    }
                }      
            }
        }
        return result;   
    }
    bool check9( vector<vector<char>>& board, int m, int n ) {
        set<char> s;
        bool result = true;
        for ( int i = m-1 ; i <= m+1 && result ; i++ ) {
            for ( int j = n-1 ; j <= n+1 && result ; j++ ) {
                if ( board[i][j] != '.' ) {
                    if ( s.find(board[i][j]) == s.end() ) {
                        s.insert( board[i][j] );
                    }
                    else {
                        result = false;
                    }
                }
            }
        }
        return result;
    }
    bool isValidSudoku(vector<vector<char>>& board) {
        bool result = true;
        for ( int i = 1 ; i <= 7 && result ; i+=3 ) {
            for ( int j = 1 ; j <= 7 && result ; j+=3 ) {
                result = check9( board, i, j );
            }
        }
        for ( int i = 0, j = 0 ; i < 9 && result ; i++, j = i ) {
            result = checkLine( board, i, j );
        }
        return result;
    }
};
```
