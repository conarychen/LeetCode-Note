# [90. Subsets II](https://leetcode.com/problems/subsets-ii/)  
可以直接看[40](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0040.%20Combination%20Sum%20II.md)  
一樣用回朔+去重找出所有子集合  
```cpp
class Solution {
private:
    vector<vector<int>> result;
    vector<int> temp;

    void findSub( vector<int>& nums, int start, vector<bool> used ) {
        if ( start > nums.size()-1 ) {
            return;
        }
        for ( int i = start ; i < nums.size() ; i++ ) {
            // 去重複 詳細在40題有註解
            // !!!!!這種解法 要先把nums排列 很重要
            if ( i > 0 && nums[i] == nums[i-1] && used[i-1] == false ) {
                continue;
            }
            temp.push_back( nums[i] );
            result.push_back( temp );
            used[i] = true;
            findSub( nums, i+1, used );
            temp.pop_back();
            used[i] = false;

        }
    }
public:
    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        result.push_back( temp );
        vector<bool> used( nums.size(), false );
        sort( nums.begin(), nums.end() );
        findSub( nums, 0, used );
        return result; 
    }
};
```
