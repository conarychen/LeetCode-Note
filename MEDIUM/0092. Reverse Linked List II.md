# [92. Reverse Linked List II](https://leetcode.com/problems/reverse-linked-list-ii/description/)  
目標 -> 把範圍left ~ right內的node反轉  
我先找到範圍的開頭和結尾以及前後( pre -> next = start end -> next = endN)
把end切斷後 將start丟入反轉的函式 把範圍內的這條反轉  
出來後若原本start是head那就把head改成end(因為已經反轉了) 否則 pre -> next = end  
而反轉後的start就接上endN 即可  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head ) {
        ListNode* temp = head,* nextN,* pre;
        if ( head == NULL ) {
            return NULL;
        }
        nextN = temp -> next;
        temp -> next = NULL;
        while ( nextN != NULL ) {
            pre = temp;
            temp = nextN;
            nextN = nextN -> next;
            temp -> next = pre;
        }

        return temp;
    }
    ListNode* reverseBetween(ListNode* head, int left, int right) {
        if ( left == right ) {
            return head;
        }

        ListNode* pre = NULL, * start = head, * end = head;
        ListNode* endN = NULL;
        bool find = false;
        int count = 1;
        
        while ( !find ) {
            
            if ( count < left ) {
                pre = start;
                start = start -> next;
            }

            if ( count < right ) {
                end = end -> next;
            }
            else {
                find = true;
            }
            count++;
        }
        if ( end != NULL ) {
            endN = end -> next;
        }
        end -> next = NULL;
        if ( left != 1 ) {
            pre -> next = NULL;
        }
        else {
            pre = NULL;
        }
        
        end = reverseList( start );
        if ( left == 1 ) {
            head = end;
        }
        else {
            pre -> next = end;
        }
        
        start -> next = endN;
        return head;
    }
};
```
