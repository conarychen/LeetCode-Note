# [216. Combination Sum III](https://leetcode.com/problems/combination-sum-iii/description/)  
目標 -> 用k個數字組合出n 每個組合內的數字不可重複使用  
利用遞迴+回朔  
每次遞迴傳入k-1當作計數 當k == 0 代表使用數量上限 就可以看看是否等於n  
```cpp
class Solution {
private:
    vector<vector<int>> result;
    vector<int> temp;
    void calculate( int k, int n, int start, int sum ) {
        if ( k == 0 ) {
            if ( sum == n ) {
                result.push_back( temp );
            }
            return;
        }
        for ( int i = start ; i < 10 ; i++ ) {
            temp.push_back( i );
            sum += i;
            calculate( k-1, n, i+1, sum );
            temp.pop_back();
            sum-= i;
        }
    }
public:
    vector<vector<int>> combinationSum3(int k, int n) {
        calculate( k, n, 1, 0 );
        return result;
    }
};
```
