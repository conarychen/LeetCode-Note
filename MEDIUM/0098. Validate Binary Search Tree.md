# [98. Validate Binary Search Tree](https://leetcode.com/problems/validate-binary-search-tree/description/)  
目標 -> 驗證Tree是否為BST  
上面的orGate完全不需要 直接用運算子(||)就好  
我用傳上下限的方式 每次經過node就檢查有沒有在上下限內  
因為怕溢位所以我用long long  
如果可以走到leaf 就代表路線上都是合規的  
左子樹low = parent的low high = parent val  
右子樹low = parent val high = parent的high  
也可以用inorder traversal(我覺得這比較好)  
走訪結果在BST一定會由小到大 所以傳上一個走過的比對有沒有變小就可以  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    bool orGate( bool a, bool b ) {
        
        if ( a == false || b == false ) {
            return false;
        }
        return true;
    }
    bool checkingBST( TreeNode* root, long long low, long long high ) {
        //每次都傳上下限來比對
        if ( root == NULL ) {
            return true;
        }
        else if ( root -> val <= low || root -> val >= high ) {
            return false;
        }
        else {
            return orGate( checkingBST( root -> left, low, root -> val ), checkingBST( root -> right, root -> val, high ) );
        }
    }
    bool isValidBST(TreeNode* root) { 
        return checkingBST( root, LONG_MIN, LONG_MAX );
        //也可以用inorder traversal
        //走訪結果在BST一定會由小到大 所以傳上一個走過的比對有沒有變小就可以
    }
};
```
