# [328. Odd Even Linked List](https://leetcode.com/problems/odd-even-linked-list/description/)  
目標 -> 把單雙數位置的node分成前後半  
這題我我覺得可以直接用兩個head 每次移動看計數出來目前是單還是雙就把目前的節點接到相應的list上  
原本這樣寫法也類似但是兩條線一起往下 會比較雜一點  
用一個pointer遍歷+counter來看要接到哪比較容易讀  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* oddEvenList(ListNode* head) {
        ListNode* result = NULL;
        if ( head != NULL && head -> next != NULL ) {
            ListNode* headO = head, * headE = head -> next;
            ListNode* tempO = headO, * tempE = headE;
            ListNode* preO = NULL, * preE = NULL;
            while ( tempO != NULL && tempE != NULL ) {
                tempO -> next = tempE -> next;
                preO = tempO;
                tempO = tempO -> next;
                if ( tempO != NULL ) {
                    tempE -> next = tempO -> next;    
                    preE = tempE;      
                    tempE = tempE -> next;
                }

            }
            if ( tempO != NULL ) {
                tempO -> next = headE;
            }
            else {
                preO -> next = headE;
            }
            result = headO;

        }
        else {
            result = head;
        }
        return result;
    }
};
```
