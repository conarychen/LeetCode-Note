```cpp
class Solution {
public:
    int matchPlayersAndTrainers(vector<int>& players, vector<int>& trainers) {
        int count = 0, num = 0;
        sort( players.begin(), players.end() );
        sort( trainers.begin(), trainers.end() );
        for ( int i = 0 ; i < trainers.size() && num < players.size() ; i++ ) {    
            if ( trainers[i] >= players[num] ) {
                count++;
                num++;
            }
        }

        return count;
    }
};
```
