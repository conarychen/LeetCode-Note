# [40. Combination Sum II](https://leetcode.com/problems/combination-sum-ii/description/)  
與[39](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0039.%20Combination%20Sum.md)相同  
差別在於39沒有重複元素 而此題有 這樣會造成可能取到重覆組合    
所以要去重複 利用vector來記錄同一輪中有用過的數字 避免重覆組合  
(註解中有解釋詳細想法)  

```cpp
class Solution {
private:
    vector<vector<int>> result;
    vector<int> temp;
    
    void counting( vector<int>& candidates, int target, int count, int start, vector<bool>& used ) {
        if ( count == target ) {
            result.push_back( temp );
            return;
        }
        else if ( count > target ) {
            return;
        }

        for ( int i = start ; i < candidates.size() ; i++ ) {
            // candidate[i] == candidate[i-1]的情況下
            // used[i-1] == true 代表是有用到重複數字的同一輪(同時用到i, i-1)
            // used[i-1] == false 則代表i-1已經用完 回歸成false
            // 所以第二種情況會導致找到一樣的結果 就把她跳掉
            if ( i > 0 && candidates[i] == candidates[i-1] && used[i-1] == false ) {
                continue;
            }

            count += candidates[i];
            temp.push_back( candidates[i] );
            used[i] = true;
            counting( candidates, target, count, i+1, used );
            count -= candidates[i];
            temp.pop_back();
            used[i] = false; 
        }

    }
public:
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        sort( candidates.begin(), candidates.end() );
        vector<bool> used( candidates.size(), false );
        counting( candidates, target, 0, 0, used );
        return result;
    }
};
```
