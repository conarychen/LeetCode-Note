# [24. Swap Nodes in Pairs](https://leetcode.com/problems/swap-nodes-in-pairs/description/)  
目標 -> 每兩個node互換  
最簡單方法是直接改值 但這題要求一定要交換節點  
所以會需要以下四個pointer來記錄node  
pre -> one -> two -> nextN  
而要把這樣的順序變成  
pre -> two -> one -> nextN  
因為同樣會動到head  
所以我用了dummy head 就不用特別把head的部分拉出來  
主要特別注意每個node or pointer指向的是否是NULL 避免操作出問題  

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    // pre -> two -> one -> nextN
    ListNode* swapPairs(ListNode* head) {
        ListNode* dummy = new ListNode(0, head);
        ListNode* pre = dummy,* nextN = NULL;
        ListNode* one,* two;
        
        if ( head != NULL ) {
            one = head;
            if ( head -> next != NULL ) {
                two = head -> next;
                if ( two -> next != NULL ) {
                    nextN = two -> next;
                }
            }
        }
        // pre -> two -> one -> nextN
        while ( one != NULL && two != NULL ) {
            pre -> next = two;
            two -> next = one;
            if ( nextN != NULL ) {
                one -> next = nextN;
                
            }
            else {
                one -> next = NULL;
                
            }
            pre = one;
            one = one -> next;
            if ( one != NULL ) {
                two = one -> next;
                if ( two != NULL ) {
                    nextN = two -> next;
                }
                else {
                    nextN = NULL;
                }
            }
            else {
                two = NULL;
            }


        }

    return dummy -> next; 
    }
    
};
```
