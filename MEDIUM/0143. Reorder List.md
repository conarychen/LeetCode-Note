# [143. Reorder List](https://leetcode.com/problems/reorder-list/description/)  
目標 -> 將linked list順序改為頭尾交錯  
Input: head = [1,2,3,4]  
Output: [1,4,2,3]  
這邊比較暴力 直接把所有node存到vecor中  
再以對的順序接起來  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    void reorderList(ListNode* head) {
        vector<ListNode*> vec;
        ListNode* dummy = new ListNode();
        dummy -> next = head;
        ListNode* temp, * back;
        bool isEnd = false;
        for ( temp = head ; temp != NULL ; temp = temp -> next ) {
            if ( temp != NULL ) {
                vec.push_back( temp );
            }
        }
        temp = dummy;
        for ( int i = 0 , j = vec.size()-1 ; !isEnd ; i++, j-- ) {
            temp -> next = vec[i];
            temp = temp -> next;
            if ( i != j ) {
                temp -> next = vec[j];
                temp = temp -> next;
            }
            else {
                isEnd = true;
                temp -> next = NULL;
            }

            if ( i+1 == j ) {
                isEnd = true;
                temp -> next = NULL;
            }
        }
        head = dummy -> next;
    }
};
```
