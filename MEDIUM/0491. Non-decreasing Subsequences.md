# [491. Non-decreasing Subsequences](https://leetcode.com/problems/non-decreasing-subsequences/description/)  
目標 -> 找出所有非遞減的子序列(序列不需要連續)長度至少要為2  
用遞迴+回朔來找組合  
每層都用set來記錄同個位置有用過那些數  
像是[3,4,7,7] -> [3,4or7] 只有兩種情況 但7有兩個所以要去重複  
temp拿來找組合 result則把符合的temp存進去當結果  
```cpp
class Solution {
private:
    vector<vector<int>> result;
    vector<int> temp;
    
    void finding( vector<int> nums, int start ) {
        if ( temp.size() >= 2 ) {
            result.push_back( temp );
        }
        if ( start >= nums.size() ) {
            return;
        }
        unordered_set<int> used;
        for ( int i = start ; i < nums.size() ; i++ ) {
            if ( (temp.size() != 0 && nums[i] < temp.back()) ||
                 ( used.find( nums[i] ) != used.end() ) ) {
                continue;
            }
            
            temp.push_back( nums[i] );
            //用set紀錄同個位置用過那些數字
            //每層分開算
            used.insert( nums[i] );
            finding( nums, i+1 );
            temp.pop_back();
        }
    }
public:
    vector<vector<int>> findSubsequences(vector<int>& nums) {
        finding( nums, 0 );
        return result;
    }
};
```
