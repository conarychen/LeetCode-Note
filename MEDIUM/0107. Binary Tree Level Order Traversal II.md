# [107. Binary Tree Level Order Traversal II](https://leetcode.com/problems/binary-tree-level-order-traversal-ii/description/)  
跟[102](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0102.%20Binary%20Tree%20Level%20Order%20Traversal.md)一樣  
只是從下往上 所以我方法用一樣 最後把結果倒序而已  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> levelOrderBottom(TreeNode* root) {
        vector<vector<int>> result;
        vector<int> level;
        vector<TreeNode*> que;
        int sizeQ;
        if ( root != NULL ) {
            que.push_back( root );
        }       
        while ( que.size() != 0 ) {
            sizeQ = que.size();
            for ( int i = 0 ; i < sizeQ ; i++ ) { //只做一開始queue裡面的 = 同一層
                level.push_back( que[0] -> val );
                if ( que[0] -> left != NULL ) { //左邊有就加進queue
                    que.push_back( que[0] -> left );
                }
                if ( que[0] -> right != NULL ) { //右邊有加進queue
                    que.push_back( que[0] -> right );
                }
                que.erase( que.begin() ); //結束後移出queue
            } //結束後裡面剩下的就是下一層的
            result.push_back( level ); //將每層結果推入result
            while ( level.size() != 0 ) { 
                level.pop_back();
            }
        } 
        // reverse result
        for ( int i = 0 ; i < result.size()/2 ; i++ ) {
            swap( result[i], result[result.size()-1-i] );
        }
        return result;
    }
};
```
