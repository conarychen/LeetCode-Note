# [86. Partition List](https://leetcode.com/problems/partition-list/description/)  
目標 -> 把val < x的node移動到list的後半部 且其餘部分要維持原順序  
我用兩個dummy head 走訪一次 把大小於拆成兩條後  
再把大於那條的尾接上另一條的頭即可  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* partition(ListNode* head, int x) {
        // two dummy head
        ListNode *dummyS = new ListNode(), *dummyB = new ListNode();
        ListNode *small = dummyS, *big = dummyB;
        for ( ListNode *temp = head ; temp != NULL ; temp = temp -> next ) {
            if ( temp -> val < x ) {

                small -> next = temp;
                small = small -> next;
            }
            else {
                big -> next = temp;
                big = big -> next;
            }
        }
        small -> next = NULL;
        big -> next = NULL;
        small -> next = dummyB -> next;
        return dummyS -> next;
    }
};
```
