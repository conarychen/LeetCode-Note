# [57. Insert Interval](https://leetcode.com/problems/insert-interval/)  
我拿[56](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0056.%20Merge%20Intervals.md)的來用  
基本上插入後就做一樣的事  
```cpp
class Solution {
public:
    // compare ther first element in vector
    static bool cmp( vector<int> &a, vector<int> &b ) {
        return a[0] < b[0];
    }
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        vector<vector<int>> result = intervals;
        // sort and compare 
        // [1,4] [4,7] -> 4(left) >= 4(right) -merge-> [1,7]
        sort( result.begin(), result.end(), cmp );
        for ( int i = 0 ; i < result.size()-1 ; i++ ) {
            if ( result[i][1] >= result[i+1][0] ) {
                result[i][1] = max( result[i][1], result[i+1][1]);
                result.erase( result.begin() + i+1 );
                i--;
            }
        }
        return result;
    }
    vector<vector<int>> insert(vector<vector<int>>& intervals, vector<int>& newInterval) {
        intervals.push_back( newInterval );
        return merge( intervals );
    }
};
```
