# [59. Spiral Matrix II](https://leetcode.com/problems/spiral-matrix-ii/)  
和[54](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0054.%20Spiral%20Matrix.md)類似  
只是這次給n 要回傳n*n的矩陣 且內容為走訪順序  
因為從1開始 所以走過的我用0標記 其餘就跟54相同  
```cpp
class Solution {
public:
    enum direction {
        right, down, left, up, end
    };
    vector<vector<int>> generateMatrix(int n) {
        vector<vector<int>> result( n, vector<int>(n,0) );
        direction dir = right;
        bool isEnd = false;
        int row = 0, col = 0;
        int maxRow = n-1, maxCol = n-1;
        int count = 1;
        while ( !isEnd ) {
            // 0 means have not gone that position it's a tag
            // if [row][col] is 0 then put in count
            if ( ( row <= maxRow && row >= 0 ) && 
                 ( col <= maxCol && col >= 0 ) && 
                   result[row][col] == 0 ) {
                result[row][col] = count;
                count++;
            }
            else { // after change position if it's not 0 
                   // it means done
                dir = end;
            }
            // follow the rule right -> down -> left -> up -> right....
            if ( dir == right ) {
                if ( col+1 > maxCol || result[row][col+1] != 0 ) {
                    dir = down;
                    row++;
                }
                else {
                    col++;
                }
            }
            else if ( dir == down ) {
                if ( row+1 > maxRow || result[row+1][col] != 0 ) {
                    dir = left;
                    col--;
                }
                else {
                    row++;
                }
            }
            else if ( dir == left ) {
                if ( col-1 < 0 || result[row][col-1] != 0 ) {
                    dir = up;
                    row--;
                }
                else {
                    col--;
                }
            }
            else if ( dir == up ) {
                if ( row-1 < 0 || result[row-1][col] != 0 ) {
                    dir = right;
                    col++;
                }
                else {
                    row--;
                }
            }
            else {

                isEnd = true;
            }
        }
        return result;
    }
};
```
