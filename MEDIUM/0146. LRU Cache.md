# [146. LRU Cache](https://leetcode.com/problems/lru-cache/description/)  
目標 -> 設計LRU Cache function  
在有限空間下進行存儲  
存的時候如果該筆資料在 就更新內容 然後把該筆資料的活躍順序拉到第一  
如果沒有 那就看空間夠不夠  
空間夠就直接放  
空間不夠 那就把活躍度最差的踢掉 改放此筆資料  
取的部分因為是用hash所以取很快 但一樣要改活躍度  
我這邊活躍度應該要再優化 花太多資源  
```cpp
class LRUCache {
public:
    vector<int> useOrder;
    unordered_map<int, int> mp;
    int mpSize;
    LRUCache(int capacity) {
        mpSize = capacity;
    }
    
    int get(int key) {
        if ( mp.find(key) != mp.end() ) {
            //搜尋key值來更新使用順序這邊可以優化
            for ( int i = 0 ; i < useOrder.size() ; i++ ) {
                if ( useOrder[i] == key ) {
                    useOrder.erase( useOrder.begin() + i );
                    useOrder.push_back( key );
                    i = useOrder.size();
                }
            }
            return mp[key];
        }

        return -1;
    }
    
    void put(int key, int value) {

        if ( mp.find( key ) != mp.end() ) { //替換值
            mp[key] = value;

            for ( int i = 0 ; i < useOrder.size() ; i++ ) {
                if ( useOrder[i] == key ) {
                    useOrder.erase( useOrder.begin() + i );
                    useOrder.push_back( key );
                    i = useOrder.size();
                }
            }
        }
        else if ( mp.size() == mpSize ) { //踢掉最不活躍的資料 換取新資料
            mp.erase(useOrder[0]);
            useOrder.erase( useOrder.begin() );
            mp[key] = value;
            useOrder.push_back( key );
        }
        else {
            mp[key] = value;
            useOrder.push_back( key );
        }

    }
};

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);
 */
```
