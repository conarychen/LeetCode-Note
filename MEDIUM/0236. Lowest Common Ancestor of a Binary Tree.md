# [236. Lowest Common Ancestor of a Binary Tree](https://leetcode.com/problems/lowest-common-ancestor-of-a-binary-tree/description/)  
與[235](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0235.%20Lowest%20Common%20Ancestor%20of%20a%20Binary%20Search%20Tree.md)差在這題只是一般的二元樹 而非有排列規則的二元搜尋樹  
這題我由下往上做  
因為BT沒有排列規則 所以只好全部走訪來找  
題目有確保p,q一定在樹中  
所以不會有找不到的問題  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
    TreeNode* dfs( TreeNode* root , TreeNode* p, TreeNode* q ) {
        if ( root == NULL ) {
            return NULL;
        }

        TreeNode* left = dfs( root -> left, p, q );
        TreeNode* right = dfs( root -> right, p, q );
        if ( root == q || root == p ) {
            return root;
        }
        else if ( left != NULL && right == NULL ) return left;
        else if ( left == NULL && right != NULL ) return right;
        else if ( left == NULL && right == NULL ) return NULL;
        else return root;
    }
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        TreeNode* result = dfs( root, p, q );
        return result;
    }
};
```
