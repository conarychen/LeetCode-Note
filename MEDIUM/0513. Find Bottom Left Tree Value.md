# [513. Find Bottom Left Tree Value](https://leetcode.com/problems/find-bottom-left-tree-value/description/)  
目標 -> 找最左下角的node val !!!深度優先 先找最深的 然後才是最左的  
我先找最大深度  
接著像[102](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0102.%20Binary%20Tree%20Level%20Order%20Traversal.md)一樣  
一層一層取 直到最後一層的第一個node就是最左下角的node了  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:
    int maxDepth(TreeNode* root) {
        if ( root == NULL ) {
            return 0;
        }
        return max( maxDepth(root -> left)+1, maxDepth(root -> right)+1 );
    }
public:
    int findBottomLeftValue(TreeNode* root) {
        queue<TreeNode*> que;
        TreeNode* temp = NULL;
        int count = 0;
        que.push( root );
        int times = maxDepth( root );
        //取完深度後 以inorder順序取子樹(左->右)
        //最後留下最後一層的首個便是最左
        for ( int j = 0 ; j < times-1 ; j++ ) {
            count = que.size();
            for ( int i = 0 ; i < count ; i++ ) {
                temp = que.front();
                que.pop();
                if ( temp -> left != NULL ) {
                   que.push( temp->left ); 
                }
                if ( temp -> right != NULL ) {
                   que.push( temp->right ); 
                }
                
            }


        }
        return que.front()->val; 
    }
};
```
