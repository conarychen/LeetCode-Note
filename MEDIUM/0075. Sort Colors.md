# [75. Sort Colors](https://leetcode.com/problems/sort-colors/description/)  
大學時有教授講過類似題目(建宏還是執彰)  
如果是元素類別不多的 直接算完後 從頭重新填進數值就好  
此題就是如此 直接計算0 1 2 的數量後  
直接重新填入即可  

```cpp
class Solution {
public:
    void sortColors(vector<int>& nums) {
        int count0 = 0, count1 = 0, count2 = 0;
        int pos = 0;
        for ( int i = 0 ; i < nums.size() ; i++ ) {
            if ( nums[i] == 0 ) {
                count0++;
            }
            else if ( nums[i] == 1 ) {
                count1++;
            }
            else if ( nums[i] == 2 ) {
                count2++;
            }
        }
        for ( int i = 0 ; i < nums.size() ; i++ ) {
            if ( count0 > 0 ) {
                nums[i] = 0;
                count0--;
            }
            else if ( count1 > 0 ) {
                nums[i] = 1;
                count1--;
            }
            else if ( count2 > 0 ) {
                nums[i] = 2;
                count2--;
            }
        }
    }
};
```
