# [103. Binary Tree Zigzag Level Order Traversal](https://leetcode.com/problems/binary-tree-zigzag-level-order-traversal/)  
和[102](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0102.%20Binary%20Tree%20Level%20Order%20Traversal.md)一樣做法  
差在每層要左->右 右->左交錯順序 用pos來記錄要哪個方向  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
        vector<vector<int>> result;
        vector<int> level;
        vector<TreeNode*> que;
        int sizeQ, pos = 0; // 0 l to r / 1 r to l
        if ( root != NULL ) {
            que.push_back( root );
        }       
        while ( que.size() != 0 ) {
            sizeQ = que.size();
            for ( int i = 0 ; i < sizeQ ; i++ ) { //只做一開始queue裡面的 = 同一層
                level.push_back( que[0] -> val );
                if ( que[0] -> left != NULL ) { //左邊有就加進queue
                    que.push_back( que[0] -> left );
                }
                if ( que[0] -> right != NULL ) { //右邊有加進queue
                    que.push_back( que[0] -> right );
                }
                que.erase( que.begin() ); //結束後移出queue
            } //結束後裡面剩下的就是下一層的
            if ( pos == 1 ) { //reverse level
                int temp;;
                for ( int i = 0 ; i < level.size()/2 ; i++ ) {
                    temp = level[i];
                    level[i] = level[level.size()-1-i];
                    level[level.size()-1-i] = temp;
                }
                pos = 0;
            }
            else {
                pos = 1;
            }
            result.push_back( level ); //將每層結果推入result
            while ( level.size() != 0 ) { 
                level.pop_back();
            }
        } 
        return result;    
    }
};
```
