# [29. Divide Two Integers]()  
目標 -> 不使用* / % 來達成除法  
首先把變數改成long long 避免計算過程溢位  
把除數 被除數都改成正數方便計算  
利用 <<, >> 移位運算 來加速運算  
以10 / 2 為例 用平方倍快速減少
10 - 2 = 8 -> 10 - 2^2 = 6 -> 10 - 2^3 = 2 -> 10 - 2^4 = -6  
(不行 為負 所以取2^3 -> count = 3)  
最後如果還有剩 就把剩下的數減去被除數的次數加回count即可得到商  

```cpp
class Solution {
public:
    int divide(int dividend, int divisor) {
        long long count = 1;
        long long divd = dividend, divi = divisor;
        bool isNeg = false;
        if ( divd == INT_MIN && divi == -1 ) {
            return INT_MAX;
        }
        else if ( divd == 0 ) {
            return 0;
        }
        else {
            
            
            long long temp, tempDiv;
            if ( (divd < 0 && divi > 0) || (divd > 0 && divi < 0) ) {
                isNeg = true;
            }
            if ( divd < 0 ) {
                divd = -divd;
            }
            if ( divi < 0 ) {
                divi = -divi;
            }
            temp = divd;
            tempDiv = divi;
            while ( temp > 0 ) {
                
                temp -= tempDiv;
                if ( temp > 0 ) {
                    temp = divd;
                    tempDiv <<= 1; // *2
                    count <<= 1; // *2
                }
            }
            if ( temp != 0 ) { //回到上一動
                count >>= 1;
                tempDiv >>= 1;
            }
            divd -= tempDiv;
            while ( divd > 0 ) {
                divd -= divi;
                if ( divd >= 0 ) {
                    count++;
                }
            }
        }

        if ( isNeg ) {
            count = -count;
        }

        return count;
    }
};
```
