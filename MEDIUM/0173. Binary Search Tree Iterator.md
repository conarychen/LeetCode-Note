# [173. Binary Search Tree Iterator](https://leetcode.com/problems/binary-search-tree-iterator/description/)  
我用stack把node存進去後維護  
基本上按照題目要的功能做沒什麼問題  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class BSTIterator {
public:
    vector<TreeNode*> stack;
    TreeNode* root;
    BSTIterator(TreeNode* root) {
        this -> root = root;
        for ( TreeNode* temp = root ; temp != NULL ; temp = temp -> left ) {
            stack.push_back( temp );
        }
    }
// 放自己以及左子樹 要左到底 (初始化時也是) 每次要取next時確認該node有沒有右子樹
// 有的話就一樣右子樹本人跟他左子樹左到底都丟進去stack
// 這樣就可以全部走完  
    int next() { 
        int result = stack[stack.size()-1] -> val;
        if ( stack[stack.size()-1] -> right != NULL ) {
            TreeNode* temp = stack[stack.size()-1] -> right;
            stack.pop_back();
            for ( ; temp != NULL ; temp = temp -> left ) {
                stack.push_back( temp );
            }
        }
        else {
            stack.pop_back();
        }
        return result;
    }
    
    bool hasNext() {
        if ( stack.size() != 0 ) {
            return true;
        }
        return false;
    }
};

/**
 * Your BSTIterator object will be instantiated and called as such:
 * BSTIterator* obj = new BSTIterator(root);
 * int param_1 = obj->next();
 * bool param_2 = obj->hasNext();
 */
```
