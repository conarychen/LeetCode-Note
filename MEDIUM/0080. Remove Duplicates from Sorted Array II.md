# [80. Remove Duplicates from Sorted Array II](https://leetcode.com/problems/remove-duplicates-from-sorted-array-ii/description/)  
目標 -> 每個元素最多只能保留2個 !!!!但只看回傳數量的元素 因此不需要真的刪除  
利用兩個pointer  
一個用來標記要填入的位置 一個用來讀數字  
詳細流程在下面註解裡  
此題重點是不要真的刪除  
利用雙指標把讀到的結果放到該放的位置並且計數即可  
第一段註解內是真的有刪除的方法 但不推薦  
```cpp
class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        /*
        int fast = 2, slow = 0;
        // ex 1 1 1 2 2 3
        //    s   f       -> s == f -> delete f
        //->  1 1 2 2 3
        //  s   f        
        //    1 1 2 2 3
        //    s   f     -> s != f -> next
        if ( nums.size() > 2 ) {
            for ( ; fast < nums.size() ; fast++, slow++ ) {
                if ( nums[fast] == nums[slow] ) {
                    nums.erase( nums.begin()+fast );
                    fast--;
                    slow--;
                }
            }
        }
        return nums.size();
        */
        int read = 0, write = 0;
        for ( ; read < nums.size() && nums.size() > 2 ; read++ ) {
            if ( read < 2 ) {
                nums[write] = nums[read];
                write++;
            }
            else {
                if ( nums[read] != nums[write-2] ) {
                    nums[write] = nums[read];
                    write++;
                }
            }
        }
        if ( write == 0 ) {
            write = nums.size();
        }
        return write;
    }
};
```
