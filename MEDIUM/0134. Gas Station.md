# [134. Gas Station](https://leetcode.com/problems/gas-station/description/)  
目標 -> 回傳從哪個加油站出發可以跑一圈回到原點  
intput -> gas = [1,2,3,4,5], cost = [3,4,5,1,2]  
這題沒什麼特別的  
就是都跑跑看 如果gas < 0 代表沒油了 那就可以跳過換下一輪  
```cpp
class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int result = -1, my = 0;
        if ( gas.size() == 1 ) {
            if ( gas[0] >= cost[0] ) {
                result = 0;
            }
        }
        else {
            for ( int i = 0 ; i < gas.size() && result == -1 ; i++ ) {
                if ( gas[i] > cost[i] ) { // start condition
                    my = gas[i]; // start 
                    int j = i;
                    do {
                        my -=cost[j]; //minus j to j+1's cost = -cost[j]
                        j++;
                        if ( j >= gas.size() ) {
                            j = 0;
                        }
                        if ( my > 0 ) {
                            my += gas[j]; // add next gas station's gas

                        }
                        
                    } while ( j != i && my > 0 );

                    if ( my >= 0 && j == i ) {

                        result = i;
                    }
                }
            }
        }

        return result;
    }
};
```
