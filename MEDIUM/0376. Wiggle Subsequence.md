# [376. Wiggle Subsequence](https://leetcode.com/problems/wiggle-subsequence/description/)  
目標 -> 找到最長的搖擺子序列(序列不需連續位置)  
簡單來說 可以把輸入看成折線圖 而我們只取有轉折的地方  
dir是目前的斜率 preDir則是上一段的  
所以只要兩個互為1 -1就代表有轉折 那這段就是我們要的 所以計數+1  
其他不符合的情況就不用動作  
```cpp
class Solution {
public:
    int wiggleMaxLength(vector<int>& nums) {
        int count = 1;
        int dir = 0; // flat = 0, up = 1, down = -1, default = -2
        int preDir = -2;
        if ( nums.size() <= 1 ) {
            return nums.size();
        }
        // dir 不需要0的部分 但我決定先這樣不亂動
        // 等下次回來重構再來改
        for ( int i = 1 ; i < nums.size() ; i++ ) {

            if ( nums[i] - nums[i-1] > 0 ) {
                dir = 1;
            }
            else if ( nums[i] - nums[i-1] < 0 ) {
                dir = -1;
            }
            else {
                continue;
            }
            if ( preDir == -2 && dir != 0 ) {
                count++;
            }
            else {
                if ( preDir != dir ) {
                    if ( (preDir == 1 && dir == -1) 
                        || (preDir == -1 && dir == 1) 
                        || ( preDir == 0 && dir != 0 ) ) {
                        count++;
                    }
                }
            }
            preDir = dir;
        }
        return count;
    }
};
```
