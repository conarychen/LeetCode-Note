# [71. Simplify Path](https://leetcode.com/problems/simplify-path/description/)  
目標 -> 將輸入按照UNIX系統的檔案路徑規則來簡化  
先把字串parse成一個一個的路徑  
ex "/home//foo/" 拆成 -> "/home", "/foo"(多斜線視為一個斜線) 
接著處理剛剛切好的token 主要是處理"/." "/.."  
如果是"/."就跳過 因為是代表目前目錄  
"/.."則是去掉前一個路徑 因為是代表要回到父路徑  
好了後檢查如果甚麼都沒剩 那就補一個"/" 代表根  
如果最後面的是"/"且長度不為1也直接去掉即可  

```cpp
class Solution {
public:
    string simplifyPath(string path) {
        vector<string> parser;
        string temp = "";
        // parse to single path
        for ( int i = 0 ; i < path.size() ; i++ ) {
            if ( temp.size() == 0 ) {
                temp.push_back( path[i] );
            }
            else {
                if ( path[i] == '/' ) {
                    if ( temp[temp.size()-1] == '/' ) {
                        ; // // -> /
                    }
                    else {
                        parser.push_back( temp );
                        temp.clear();
                        temp.push_back( path[i] );
                    }
                }
                else {
                    temp.push_back( path[i] );
                }
            }
        }
        if ( temp.size() != 0 ) {
            parser.push_back( temp );
        }
        // process to simplified path
        for ( int i = 0 ; i < parser.size() ; i++ ) {
            //cout << "the word = " << parser[i] << endl;
            if ( parser[i].compare( "/.." ) == 0 ) {
                parser.erase( parser.begin()+i );
                
                if ( i != 0 ) {
                    parser.erase( parser.begin()+i-1 );
                    i--;
                }
                i--;
                
            }
            else if ( parser[i].compare( "/." ) == 0 ) {
                if ( parser.size() != 1 ) {
                    parser.erase( parser.begin()+i );
                    i--;
                }
                else {
                    parser.pop_back();
                }

            }
            
        }
        
        if ( parser.size() == 0 ) {
            parser.push_back( "/" );
        }
        else if ( parser[parser.size()-1] == "/" && parser.size() != 1 ) {
            parser.pop_back();
        }
        temp.clear();
        for ( int i = 0 ; i < parser.size() ; i++ ) {
            temp += parser[i];
        }
        return temp;
    }
};
```
