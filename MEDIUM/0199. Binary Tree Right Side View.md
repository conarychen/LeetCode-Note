# [199. Binary Tree Right Side View](https://leetcode.com/problems/binary-tree-right-side-view/description/)  
目標 -> 從BT右側看過去能看到的 也就是每一層最右邊的節點  
基本上也是[102](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0102.%20Binary%20Tree%20Level%20Order%20Traversal.md)作法  
只是每次先存每層的最後一個即可  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        vector<int> result;
        vector<TreeNode*> temp;
        int count = 0;
        if ( root != NULL ) {
            temp.push_back( root );
        }
        // 每次取一層 類似inorder順序 所以順序會是左到右
        // 因此右視角只會看到最後一個 把子樹推進vector後 父節點就可以移除
        while ( temp.size() != 0 ) {
            count = temp.size();
            result.push_back( temp[temp.size()-1] -> val );
            for ( int i = 0 ; i < count ; i++ ) {
                if ( temp[0] -> left != NULL ) {
                    temp.push_back( temp[0] -> left );
                }
                if ( temp[0] -> right != NULL ) {
                    temp.push_back( temp[0] -> right );
                }
                temp.erase( temp.begin() );
            }
            
        }
        return result;
    }
};
```
