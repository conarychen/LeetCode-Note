# [73. Set Matrix Zeroes](https://leetcode.com/problems/set-matrix-zeroes/)  
我利用第0列和第0行來記錄該行or列是否整行要為0  
ex 若matrix[2][3] == 0 -> matrix[0][3] = 0, matrix[2][0] = 0  
最後只要走訪0列0行就可以把所有要變成零的都改好  
另外若是0行0列的位置上本身為零 要額外記 在最後一次改  
避免動到紀錄的結果  
```cpp
class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        bool colZero = false, rowZero = false;
        for ( int i = 0 ; i < matrix.size() ; i++ ) {
            for ( int j = 0 ; j < matrix[0].size() ; j++ ) {
                if ( matrix[i][j] == 0 ) {
                    matrix[i][0] = 0;
                    matrix[0][j] = 0;
                    if ( i == 0 ) {
                        rowZero = true;
                        cout << "first\n";
                    }
                    if ( j == 0 ) {
                        colZero = true;
                        cout << "sec\n";
                    }
                }
            }
        }
        for ( int i = 1 ; i < matrix[0].size() ; i++ ) { //check first col
            if ( matrix[0][i] == 0 ) {
                for ( int j = 0 ; j < matrix.size() ; j++ ) {
                    matrix[j][i] = 0;
                }
            }
        }
        for ( int i = 1 ; i < matrix.size() ; i++ ) { // check first row
            if ( matrix[i][0] == 0 ) {
                for ( int j = 0 ; j < matrix[0].size() ; j++ ) {
                    matrix[i][j] = 0;
                }
            }
        }
        if ( colZero ) { // col 0
            for ( int i = 0 ; i < matrix.size() ; i++ ) {
                matrix[i][0] = 0;
            }

        }
        if ( rowZero ) { // row 0
            for ( int i = 0 ; i < matrix[0].size() ; i++ ) {
                matrix[0][i] = 0;
            }
        }

    }
};
```
