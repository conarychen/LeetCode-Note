# [78. Subsets](https://leetcode.com/problems/subsets/)  
與[77](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0077.%20Combinations.md)差不多  
這邊記得空集合一樣是子集 所以一開始先放個空集合進結果  
接著一樣利用backtracking 控制start的位置即可避免重複(一樣元素是unique的)  
```cpp
class Solution {
private:
    vector<vector<int>> result;
    vector<int> temp;
    void findSub( vector<int>& nums, int start ) {
        if ( start > nums.size()-1 ) {
            return;
        }
        for ( int i = start ; i < nums.size() ; i++ ) {

            temp.push_back( nums[i] );
            result.push_back( temp );
            findSub( nums, i+1 );
            temp.pop_back();
        }
    }
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        result.push_back( temp );
        findSub( nums, 0 );
        return result;
    }
};
```
