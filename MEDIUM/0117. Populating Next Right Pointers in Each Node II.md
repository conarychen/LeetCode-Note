# [117. Populating Next Right Pointers in Each Node II](https://leetcode.com/problems/populating-next-right-pointers-in-each-node-ii/description/)  
與[116](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0116.%20Populating%20Next%20Right%20Pointers%20in%20Each%20Node.md)一樣  
差在116是完全樹 此題則是普通的二元樹  
主要只差在會有沒有子樹的情況 但作法上完全沒差 所以用一樣的  
```cpp
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* left;
    Node* right;
    Node* next;

    Node() : val(0), left(NULL), right(NULL), next(NULL) {}

    Node(int _val) : val(_val), left(NULL), right(NULL), next(NULL) {}

    Node(int _val, Node* _left, Node* _right, Node* _next)
        : val(_val), left(_left), right(_right), next(_next) {}
};
*/

class Solution {
public:
    Node* connect(Node* root) {
        int sizeQ;
        vector<Node*> queue;
        if ( root != NULL ) {
            queue.push_back( root );
        }
        
        while ( queue.size() != 0 ) {
            sizeQ = queue.size();
            for ( int i = 0 ; i < sizeQ ; i++ ) {
                if ( i == sizeQ-1 ) {
                    queue[i] -> next = NULL;
                }
                else {
                    queue[i] -> next = queue[i+1];
                }
                if ( queue[i] -> left != NULL ) {
                    queue.push_back( queue[i] -> left );
                }
                if ( queue[i] -> right != NULL ) {
                    queue.push_back( queue[i] -> right );
                }
            }
            for ( int i = 0 ; i < sizeQ ; i++ ) {
                queue.erase( queue.begin() );
            }
        }
        return root;
    }
};
```
