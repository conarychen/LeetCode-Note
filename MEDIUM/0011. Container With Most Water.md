# [11. Container With Most Water](https://leetcode.com/problems/container-with-most-water/description/)  
min max不需要寫 STL有  
two pointer來從左右往中間找最大  
left right取小的往中間走  
每次走完都要算目前的面積有沒有大於maxW  

```cpp
class Solution {
public:
    int min( int a, int b ) {
        if ( a < b ) {
            return a;
        }
        return b;
    }
    int max( int a, int b ) {
        if ( a > b ) {
            return a;
        }
        return b;
    }
    int maxArea(vector<int>& height) {
        int maxW = 0, temp;
        int left = 0, right = height.size()-1;
        while ( left != right ) {
            temp = ( right - left )*min( height[right], height[left] );
            maxW = max( maxW, temp );
            if ( height[left] < height[right] ) {
                left++;
            }
            else {
                right--;
            }
        }
        return maxW;
    }
};
```
