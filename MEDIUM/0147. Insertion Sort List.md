# [147. Insertion Sort List](https://leetcode.com/problems/insertion-sort-list/)  
目標 -> 用insertion sort 排序list  
沒什麼特別 就走訪list一次 以插入排序的規則排放就好  

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* insertionSortList(ListNode* head) {
        ListNode* temp = head, * pos = head, * pre = head;
        for ( ; pos != NULL ; pos = pos -> next ) {
            if ( pos == head ) {
                ;// do nothing
            }
            else {
                ListNode* temPre = NULL, * target = pos; 
                bool isDone = false;
                for ( temp = head ; !isDone ; temp = temp -> next ) {
                    
                    if ( target -> val < temp -> val ) {
                        pre -> next = pos -> next;
                        pos = pre;
                        target -> next = temp;
                        if ( temp == head ) { //代表現在是頭
                            head = target;
                        }
                        else {
                            temPre -> next = target;
                        }
                        isDone = true;

                    }
                    else {
                        if ( temp == pos ) {
                            isDone = true;
                        }
                        temPre = temp;
                    }
                }

            }
            pre = pos;
        }
        return head;
    }
};
```
