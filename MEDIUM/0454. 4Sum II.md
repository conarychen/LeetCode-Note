# [454. 4Sum II](https://leetcode.com/problems/4sum-ii/description/)  
目標 -> 從input的4個vector中各取一數使其總和為零 回傳有幾種組合  
用map紀錄nums1+num2的各種組合(用set也可以)  
再來用nums3+nums4的負數來找 如果可以在map中找到 就代表這兩組加在一起為0  
```cpp
class Solution {
public:
    int fourSumCount(vector<int>& nums1, vector<int>& nums2, vector<int>& nums3, vector<int>& nums4) {
        unordered_map<int, int> map;
        for ( int i = 0 ; i < nums1.size() ; i++ ) {
            for ( int j = 0 ; j < nums2.size() ; j++ ) {
                if ( map.find( nums1[i] + nums2[j] ) != map.end() ) {
                    map[nums1[i]+nums2[j]]++;
                }
                else {
                    map[nums1[i]+nums2[j]] = 1;
                }
            }
        }
        int count = 0;
        for ( int i = 0 ; i < nums3.size() ; i++ ) {
            for ( int j = 0 ; j < nums4.size() ; j++ ) {
                if ( map.find( -(nums3[i] + nums4[j]) ) != map.end() ) {
                    count += map[-(nums3[i] + nums4[j])];
                }

            }
        }
        return count;
    }
};
```
