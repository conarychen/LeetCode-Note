# [74. Search a 2D Matrix](https://leetcode.com/problems/search-a-2d-matrix/description/)  
目標 -> 在input的matrix中能否找到target  
因為每一列都是遞增 且每列頭比上列尾大  
所以先利用每列範圍來找target會在哪列  
如果沒找到相應的範圍 那就代表沒有  
有的話就到該列找(可以用binary search比較快)  
```cpp
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
       bool result = false, isEnd = false;
       int row = 0;
       bool findInterval = false;
       while( !isEnd ) {
            
            for ( int i = 0 ; i < matrix.size() && !findInterval ; i++ ) { //先抓哪一列
                if ( target >= matrix[i][0] && target <= matrix[i][matrix[i].size()-1] ) {
                    row = i;
                    findInterval = true;
                }
            }
            if ( findInterval ) { //在該列找
                for ( int i = 0 ; i < matrix[row].size() && !isEnd ; i++ ) {
                    if ( target == matrix[row][i] ) { //一個一個找
                        result = true;
                        isEnd = true; //如果找到可以提前跳離
                    }
                }
                isEnd = true; //避免沒找到所以要設定
            }
            else { //一開始就沒找到區間代表根本沒有
                isEnd = true;
                result = false;
            }
       } 
       return result;
    }
};
```
