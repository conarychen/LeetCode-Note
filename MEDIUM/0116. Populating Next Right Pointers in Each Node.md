# [116. Populating Next Right Pointers in Each Node](https://leetcode.com/problems/populating-next-right-pointers-in-each-node/description/)  
目標 -> 填上node內的next 將每層像是linked list一樣串好

也是和[102](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0102.%20Binary%20Tree%20Level%20Order%20Traversal.md)一樣  
每層加上串起來的動作而已  

```cpp
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* left;
    Node* right;
    Node* next;

    Node() : val(0), left(NULL), right(NULL), next(NULL) {}

    Node(int _val) : val(_val), left(NULL), right(NULL), next(NULL) {}

    Node(int _val, Node* _left, Node* _right, Node* _next)
        : val(_val), left(_left), right(_right), next(_next) {}
};
*/

class Solution {
public:
    Node* connect(Node* root) {
        int sizeQ;
        vector<Node*> queue;
        if ( root != NULL ) {
            queue.push_back( root );
        }
        
        while ( queue.size() != 0 ) {
            sizeQ = queue.size();
            for ( int i = 0 ; i < sizeQ ; i++ ) {
                if ( i == sizeQ-1 ) {
                    queue[i] -> next = NULL;
                }
                else {
                    queue[i] -> next = queue[i+1];
                }
                if ( queue[i] -> left != NULL ) {
                    queue.push_back( queue[i] -> left );
                }
                if ( queue[i] -> right != NULL ) {
                    queue.push_back( queue[i] -> right );
                }
            }
            for ( int i = 0 ; i < sizeQ ; i++ ) {
                queue.erase( queue.begin() );
            }
        }
        return root;
    }
};
```
