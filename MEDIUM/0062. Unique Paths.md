# [62. Unique Paths](https://leetcode.com/problems/unique-paths/description/)  
目標 -> 從左上走到右下  
利用DP建立地圖 初始化第0行0列都設成1 代表有1種方法可以走到該格(因為只能往右或往下)  
接著能到每個的方法就是該格的上面+左邊即可  
```cpp
class Solution {
public:

    int uniquePaths(int m, int n) {
        vector<vector<int>> DP;
        
        for ( int i = 0 ; i < m ; i++ ) {
            vector<int> temp;
            for ( int j = 0 ; j < n ; j++ ) {
                if ( i == 0 || j == 0 ) {
                    temp.push_back(1);
                }
                else {
                    temp.push_back(0);
                }
                
            }
            DP.push_back(temp);
        } // making DP map 
          // ex 111     111
          //    100 ->  123
          //    100     136
        
        if ( m != 1 && n != 1 ) {
            for ( int i = 1 ; i < m ; i++ ) {
                
                for ( int j = 1 ; j < n ; j++ ) {
                    DP[i][j] = DP[i-1][j] + DP[i][j-1];
                }
                
            }
            return DP[m-1][n-1];
        }
        else {
            return 1;
        }

    }
};
```
