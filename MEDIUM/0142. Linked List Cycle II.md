# [142. Linked List Cycle II](https://leetcode.com/problems/linked-list-cycle-ii/description/)  
目標 -> 找出linked list中的迴圈起點  
利用快慢指標 快指標每次走2步 慢指標走1步  
這樣走可以找到list的中點  
但在有迴圈的情況下有個特性  
兩個指標碰頭後 將其中一個移到list的開頭 接著都走一步  
再次碰面就會是迴圈的開頭  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        // slow step 1 fast step 2 -> when they meet
        // move one of them to head and go one step per time
        // they'll meet again at the cycle entrance
        ListNode* slow = head, * fast = head, * result;
        if ( head == NULL ) {
            return NULL;
        }
        do {
            slow = slow -> next;
            if ( fast -> next != NULL ) {
                fast = fast -> next -> next;
            }
            else {
                fast = NULL;
            }
        } while ( slow != fast && fast != NULL );
        if ( fast == NULL ) {
            result = NULL;
        }
        else {
            slow = head;
            while ( slow != fast ) {
                slow = slow -> next;
                fast = fast -> next;
            }
            result = slow;
        }
        return result;
    }
};
```
