# [18. 4Sum](https://leetcode.com/problems/4sum/description/)
目標 -> 找出四個數的和為target的所有組合(不重複)  
與3Sum差不多 一樣是利用two pointers進行走訪  
差別是這裡要固定兩個數加上兩個pointers  
即 1(i) 2(j) 3(pOne) 4 5 6(pTwo)  
pOne pTwo 會依照出來的結果向中間前進  
若和大於target -> pTwo-1  小於target -> pOne+1  等於 -> 把組合存起 pOne+1 pTwo-1並行  
同樣也要去掉相同組合  
因此i, j, pOne, pTwo 每次移動都要檢查跟上個值有沒有相同  
i, j 若相同就跳過那輪迴圈(continue)  


```cpp
class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> result;
        if ( nums.size() < 4 ) {
            return result;
        } 
        int pOne = 0, pTwo = 0;
        long long count = 0;
        bool sameF = false, sameS = false;
        sort( nums.begin(), nums.end() );
        for ( int i = 0 ; i < nums.size()-3 ; i++ ) {
            if ( i > 0 && nums[i] == nums[i-1] ) {
                continue;
            }
            count = nums[i];
            for ( int j = i+1 ; j < nums.size()-2 ; j++ ) {
                if ( j > i+1 && nums[j] == nums[j-1] ) {
                    continue;
                }
                pOne = j+1;
                pTwo = nums.size()-1;
                while ( pOne < pTwo ) {
                    if ( count+nums[j]+nums[pOne]+nums[pTwo] == target ) {
                        vector<int> temp;
                        temp.push_back( nums[i] );
                        temp.push_back( nums[j] );
                        temp.push_back( nums[pOne] );
                        temp.push_back( nums[pTwo] );
                        result.push_back( temp );
                        pOne++;
                        while( pOne < nums.size() && nums[pOne] == nums[pOne-1] ) {
                            pOne++;
                        }
                        pTwo--;
                        while( pTwo > 0 && nums[pTwo] == nums[pTwo+1] ) {
                            pTwo--;
                        }
                    }
                    else if ( count+nums[j]+nums[pOne]+nums[pTwo] < target ) {
                        pOne++;
                        while( pOne < nums.size() && nums[pOne] == nums[pOne-1] ) {
                            pOne++;
                        }
                    }
                    else {
                        pTwo--;
                        while( pTwo > 0 && nums[pTwo] == nums[pTwo+1] ) {
                            pTwo--;
                        } 
                    }
                }
            }
        }
        return result;
    }
};
```
