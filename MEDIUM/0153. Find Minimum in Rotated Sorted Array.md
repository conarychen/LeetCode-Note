# [153. Find Minimum in Rotated Sorted Array](https://leetcode.com/problems/find-minimum-in-rotated-sorted-array/description/)  
目標 -> 在旋轉過的排序數組中找最小值  
簡單來說就是找pivot  
有找到就回傳該位置的元素  
沒找到代表沒旋轉 所以最小值在index 0  
```cpp
class Solution {
public:
    int findMin(vector<int>& nums) {
        int left = 0, right = nums.size()-1, mid = 0;
        int result = 0, pivot = -1;
        bool isEnd = false;


        if ( nums.size() == 1 ) {
            ;
        }
        else {
            //find pivot  ex 34512 -> my pivot means 4th place = 1
            for ( int i = 0 ; i+1 < nums.size() ; i++ ) {
                if ( nums.size() > 1 && nums[i] > nums[i+1] ) {
                    pivot = i+1; 
                    i = nums.size();
                    //find first one
                }
            }
        }

        if ( pivot != -1 ) {
            result = nums[pivot];
        }
        else {
            result = nums[0];
        }

        return result;
    }
};
```
