# [337. House Robber III](https://leetcode.com/problems/house-robber-iii/)  
和前面的不同 這題房屋排列變成樹狀  
這題的也不向之前是用member變數的DP 而是靠每層遞迴回傳的來運算後再往上傳  
利用postorder由下往回做 分成兩種狀況  
一 偷root -> 不能偷左右子  
二 不偷root -> 偷左右子的最大(代表可偷可不偷)  
回傳的vector結構是 [0] = 情況二 [1] = 情況一  
最後比對誰大選誰即可  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:
    vector<int> travel( TreeNode* root ) {
        vector<int> result( 2, 0 );
        if ( root == NULL ) return result;
        vector<int> left = travel( root -> left );
        vector<int> right = travel( root -> right );
        //steal root
        int valR = root -> val + left[0] + right[0]; 
        //don't steal root
        int valC = max( left[0], left[1] ) + max( right[0], right[1] );
        result[0] = valC;
        result[1] = valR;
        return result;
        
    }
public:
    int rob(TreeNode* root) {
        vector<int> result = travel( root );
        return result[0] > result[1] ? result[0] : result[1];
    }
};
```
