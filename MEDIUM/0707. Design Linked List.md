# [707. Design Linked List](https://leetcode.com/problems/design-linked-list/description/)  
目標 -> 按照題目寫一個linked list  
沒什麼特別 就照spec寫 都是基礎操作  
```cpp
class MyLinkedList {
public:
    struct myNode {
        int val;
        myNode *next = NULL;

        myNode( int valIn ) {
            val = valIn;
        }
    };

    int listLen;
    myNode *head = NULL;
    myNode *tail = NULL;

    MyLinkedList() {
        head = NULL;
        tail = NULL;
        listLen = 0;
    }

    
    int get(int index) {
        if ( index > listLen-1 || index < 0 ) {
            return -1;
        }
        else {
            myNode *temp = head; 
            for ( int i = 0 ; i < index ; i++ ) {
                temp = temp -> next;
            }
            return temp -> val;
        }
    }
    
    void addAtHead(int val) {
        if ( head == NULL ) {
            head = new myNode(val);
            tail = head;
        }
        else {
            myNode *temp = new myNode(val);
            temp -> next = head;
            head = temp;
        }
        listLen++;
    }
    
    void addAtTail(int val) {
        myNode *temp = new myNode(val);
        if ( head == NULL ) {
            head = temp;
            tail = temp;
        }
        else {
            tail -> next = temp;
            tail = tail -> next;
        }
        listLen++;
    }
    
    void addAtIndex(int index, int val) {
        if ( index > listLen ) {
            return;
        }
        else if ( index == 0 ) {
            addAtHead(val);
            listLen--;
        }
        else if ( index == listLen ) {
            myNode *tar = new myNode(val);
            tail -> next = tar;
            tail = tail -> next;
        }
        else {
            myNode *temp = head;
            for ( int i = 0 ; i < index-1 ; i++ ) {
                temp = temp -> next;
            }
            myNode *tar = new myNode(val);
            tar -> next = temp -> next;
            temp -> next = tar;
        }
        listLen++;
    }
    
    void deleteAtIndex(int index) {
        if ( index > listLen-1 || index < 0 ) {
            return;
        }
        else if ( index == 0 ) {
            if ( listLen == 1 ) {
                delete head;
                head = NULL;
                tail = NULL;
            }
            else {
                myNode *temp = head;
                head = head -> next; 
                delete temp;
                
            }
        }
        else {
            myNode *temp = head;
            for ( int i = 0 ; i < index-1 ; i++ ) {
                temp = temp -> next;
            }
            myNode *tar = temp -> next;
            temp -> next = tar -> next;
            if ( tar == tail ) {
                tail = temp;
            }
            delete tar;
        }
        listLen--;
    }
};

/**
 * Your MyLinkedList object will be instantiated and called as such:
 * MyLinkedList* obj = new MyLinkedList();
 * int param_1 = obj->get(index);
 * obj->addAtHead(val);
 * obj->addAtTail(val);
 * obj->addAtIndex(index,val);
 * obj->deleteAtIndex(index);
 */
```
