# [81. Search in Rotated Sorted Array II](https://leetcode.com/problems/search-in-rotated-sorted-array-ii/description/)  
與[33](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0033.%20Search%20in%20Rotated%20Sorted%20Array.md)差在 此題會有重複元素  
但我自己寫起來沒什麼差 一樣找pivot 然後binary search  
```cpp
class Solution {
public:
// 應該有地方沒寫好 因為我直接改33題的code 不覺得有動到甚麼邏輯
    bool search(vector<int>& nums, int target) {
        int left = 0, right = nums.size()-1, mid = 0;
        int pivot = 0;
        bool isEnd = false, result = false;


        if ( nums.size() == 1 ) {

            if ( target == nums[0] ) {
                result = true;
            }
            else {
                result = false;
            }
        }
        else {
            //find pivot
            for ( int i = 0 ; i+1 < nums.size() ; i++ ) {
                if ( nums.size() > 1 && nums[i] > nums[i+1] ) {
                    pivot = i+1; 
                    i = nums.size();
                    //find first one
                }
            }

            // find target,  if > nums[1] -> left part(0~pivot-1) else right part
            if ( pivot != 0 ) {
                if ( target >= nums[0] ) {
                    left = 0;
                    right = pivot-1;
                }
                else {
                    left = pivot;
                    right = nums.size()-1;
                }
            }
            else { // rotate did not occur
                left = 0;
                right = nums.size()-1;
            }

            
            while ( !isEnd && left <= right ) { //binary search for target
                
                mid = left + ( right - left ) / 2;
                
                if ( target == nums[left] || target == nums[right] ) {
                    
                    result = true;
                    isEnd = true;
                }
                else { 
                    if ( target < nums[mid] ) {
                        right = mid - 1;
                    }
                        else if ( target > nums[mid] ) {
                        left = mid + 1;
                    }
                    else {
                        result = true;
                        isEnd = true;
                    }
                }

            }
            if ( isEnd == false ) {
                result = false;
            }
            
        }

        return result;
    }
};
```
