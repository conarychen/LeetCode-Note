# [54. Spiral Matrix](https://leetcode.com/problems/spiral-matrix/description/)  
<img width="239" height="241" alt="image" src="https://github.com/user-attachments/assets/da8f9c7e-ac1d-481c-915c-d53d6a18006c" />  
目標 -> 根據傳進的陣列 從[0][0]開始以螺旋方式走訪  
因為數字範圍是-100~100 所以我把走過的格子設成-111  
接著就照著螺旋的方向(右下左上) 如果下一步會超出邊界或是下一步是走過的格子就改方向  
否則就一直前進  

```cpp
class Solution {
public:
    enum direction {
        right, down, left, up, end
    };


    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        vector<int> result;
        direction dir = right;
        bool isEnd = false;
        int row = 0, col = 0;
        int maxRow = matrix.size()-1, maxCol = matrix[0].size()-1;

        while ( !isEnd ) {
            // -111 means have gone that position it's a tag
            // if [row][col] is new one then record it
            if ( ( row <= maxRow && row >= 0 ) && 
                 ( col <= maxCol && col >= 0 ) && 
                   matrix[row][col] != -111 ) {
                result.push_back( matrix[row][col] );
                
                matrix[row][col] = -111; 
            }
            else { // after change position if it's -111 
                   // it means done
                dir = end;
            }
            // follow the rule right -> down -> left -> up -> right....
            if ( dir == right ) {
                if ( col+1 > maxCol || matrix[row][col+1] == -111 ) {
                    dir = down;
                    row++;
                }
                else {
                    col++;
                }
            }
            else if ( dir == down ) {
                if ( row+1 > maxRow || matrix[row+1][col] == -111 ) {
                    dir = left;
                    col--;
                }
                else {
                    row++;
                }
            }
            else if ( dir == left ) {
                if ( col-1 < 0 || matrix[row][col-1] == -111 ) {
                    dir = up;
                    row--;
                }
                else {
                    col--;
                }
            }
            else if ( dir == up ) {
                if ( row-1 < 0 || matrix[row-1][col] == -111 ) {
                    dir = right;
                    col++;
                }
                else {
                    row--;
                }
            }
            else {

                isEnd = true;
            }
        }
        return result;
    }
};
```
