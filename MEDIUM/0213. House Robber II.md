# [213. House Robber II](https://leetcode.com/problems/house-robber-ii/description/)  
目標 -> 和[198](https://github.com/conarychen/LeetCode-Note/blob/main/MEDIUM/0198.%20House%20Robber.md)一樣  
只是此題房屋變成圓環狀 所以頭尾相連  
這邊嘗試改成一維DP  
因為這題頭尾也相臨  
所以我改成分兩種情況最後比大小  
一是不偷第一間的情況 -> 最後一間可偷可不偷取大  
二是第一間可偷可不偷取大 -> 最後一間不偷  
最後兩種情況取大 即可  
```cpp
class Solution {
private:
    int robing(vector<int>& nums, int start, int end ) {
        int size = end-start+1;
        if ( start == end ) return nums[start];

        vector<int> dp( nums.size(), 0);
        dp[start] = nums[start];
        dp[start+1] = max(nums[start+1], dp[start]);
        for ( int i = start+2 ; i <= end ; i++ ) {
            dp[i] = max( dp[i-2]+nums[i], dp[i-1] );
        }
        return dp[end]; 
    }
public:
    int rob(vector<int>& nums) {
        if ( nums.size() == 0) return 0;
        else if ( nums.size() == 1 ) return nums[0];
        int a = robing( nums, 1, nums.size()-1 );
        int b = robing( nums, 0, nums.size()-2 );
        return a > b ? a : b;
    }
};
```
