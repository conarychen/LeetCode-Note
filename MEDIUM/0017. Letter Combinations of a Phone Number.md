# [17. Letter Combinations of a Phone Number] (https://leetcode.com/problems/letter-combinations-of-a-phone-number/description/)
目標 -> 模擬舊式手機的打字 依照輸入找出所有可能的文字組合  
與現實不同的是假設輸入為2 那輸出可以是d, e or f  
先把每個數字代表的字母建成表  
再利用遞迴進行回朔  
以23為例  
temp ==> a --> a, d --> a --> a, e --> a --> a, f --> a --> b --> b, d --> b, e --> b --> b, f --> b --> c ...
```cpp
class Solution {
private:
    vector<string> result;
    string temp;
    string map[10] = {
        "",
        "",
        "abc", //2
        "def", //3
        "ghi", //4
        "jkl", //5
        "mno", //6
        "pqrs", //7
        "tuv", //8
        "wxyz", //9
    };
    void typing( string digits, int start ) {
        if ( start == digits.size() ) {    
            result.push_back( temp );
            return;
        }
        int index = digits[start] - '0'; //把index轉成int取出
        for ( int i = 0 ; i < map[index].size() ; i++ ) {
            temp += map[index][i];
            
            typing( digits, start+1 );
            temp.pop_back();
        }
    }
public:
    vector<string> letterCombinations(string digits) {
        if ( digits.size() == 0 ) {
            return result;
        }
        typing( digits, 0 );
        return result;
    }
};
```
