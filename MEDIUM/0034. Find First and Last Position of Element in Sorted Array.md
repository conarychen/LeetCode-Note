# [34. Find First and Last Position of Element in Sorted Array](https://leetcode.com/problems/find-first-and-last-position-of-element-in-sorted-array/description/)  
目標 -> 找到target在nums中的範圍
ex 5,7,8,8,10  target = 8  output = [2,3]  
如果沒找到就[-1,-1]  
先用binary search找target  
找到後再前後找範圍即可  

```cpp
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int> res; //有點亂 因為拿前面題目改的
        int left = 0, right = nums.size()-1, mid = 0;
        int result = -1;
        bool isEnd = false;
        while ( !isEnd && left <= right ) { //logn找target
            mid = left + ( right - left ) / 2;
            
            if ( target < nums[mid] ) {
                right = mid - 1;
            }
            else if ( target > nums[mid] ) {
                left = mid + 1;
            }
            else {
                result = mid;
                isEnd = true;
            }
        }
        int i = 0;
        
        if ( result != -1 && nums.size() != 1 ) { //找到後 前後找範圍
            for ( i = result ; i > -1 && nums[i] == target ; i-- ) {
                ;
            }
            if ( i == 0 && nums[i] == target ) {
                res.push_back( i ); 
            }
            else {
                res.push_back( i+1 );
            }
            i = 0;
            for ( i = result+1 ; i < nums.size() && nums[i] == target ; i++ ) {
                ;
            }
            res.push_back( i-1 );
            
        }
        else if ( nums.size() != 1 ) {
            res.push_back( -1 );
            res.push_back( -1 );
        }
        else {
            if ( nums[0] == target ) {
                res.push_back( 0 );
                res.push_back( 0 );
            }
            else {
                res.push_back( -1 );
                res.push_back( -1 );
            }
        }
        return res;
    }
};
```
