# [160. Intersection of Two Linked Lists](https://leetcode.com/problems/intersection-of-two-linked-lists/description/)  
目標 -> listA 和 listB在某點相交變成一條list 找到相交點  
先計算兩條list的長度  
接著要把兩條list尾端對齊 也就是讓長的那條先走他多出來的數量  
這樣他們後續長度一致 只要每次都走一步 一定會從某一次開始就都是走同個node  
ex listA -> 5 listB -> 7 ---> listA從第一個node 開始 listB從第三個node開始 他們每次走一格 會一起到尾端  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        int numsA = 0, numsB = 0;
        for ( ListNode* temp = headA ; temp != NULL ; temp = temp -> next ) {
            numsA++;
        }
        for ( ListNode* temp = headB ; temp != NULL ; temp = temp -> next ) {
            numsB++;
        }
        ListNode* nodeA = headA;
        ListNode* nodeB = headB;
        if ( numsA > numsB ) {
            for ( int i = 0 ; i < numsA-numsB ; i++ ) {
                nodeA = nodeA -> next;
            } 
        }
        else if ( numsA < numsB ) {
            for ( int i = 0 ; i < numsB-numsA ; i++ ) {
                nodeB = nodeB -> next;
            } 
        }
        while ( nodeA != NULL ) {
            if ( nodeA == nodeB ) return nodeA;
            nodeA = nodeA -> next;
            nodeB = nodeB -> next;
        }
        return NULL;
    }
};
```
