# [112. Path Sum](https://leetcode.com/problems/path-sum/description/)  
目標 -> 將走過的節點內val加總 看能不能找到等於targetSum的路徑(root -> leaf)  
每走一個node就把targetSum減去目前node的val 然後傳下去  
直到leaf看是否為0 左右只要有一個就代表有路 所以用or  

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:

public:
    bool hasPathSum(TreeNode* root, int targetSum) {
        if ( root == NULL ) {
            return false;
        }
        else if ( root -> left == NULL && root -> right == NULL && targetSum-root->val == 0 ) {
            return true;
        }
        //root -> leaves 每走一個node就扣掉自己看一下有沒有到0
        //如果到葉節點的同時targetSum歸零 就代表至少有一條可以
        bool leftB = hasPathSum( root->left, targetSum-root->val );
        bool rightB = hasPathSum( root->right, targetSum-root->val );
        

        return leftB || rightB; 
    }
};
```
