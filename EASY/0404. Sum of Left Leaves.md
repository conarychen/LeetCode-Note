# [404. Sum of Left Leaves](https://leetcode.com/problems/sum-of-left-leaves/description/)  
目標 -> 計算左葉節點Val的總和  
詳細想法在下面註解  
反正就是postorder遞迴找leaf  
如果左子樹是leaf那就可以加到count  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:
    int adding( TreeNode* root, int &count ) {
        if ( root == NULL ) {
            return 0;
        }
        // find the leaves
        int left = adding( root -> left, count );
        int right = adding( root -> right, count );
        // if we got left and right both equal 0
        // means this node is a leaf node
        // then we return 1
        // so when we got return left equals 1
        // we add the value of the left node into count
        // else return -1 to do nothing
        if ( left == 1 ) {
            count += root->left->val;
        }
        else if ( left == 0 && right == 0 ) {
            return 1;
        }
        return -1;
    }
public:
    int sumOfLeftLeaves(TreeNode* root) {
        int count = 0;
        adding( root, count );
        return count;
    }
};
```
