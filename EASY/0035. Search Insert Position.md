# [35. Search Insert Position](https://leetcode.com/problems/search-insert-position/description/)  
目標 -> 尋找target可以插入到nums的哪個位置(案排序)  
用binary search找  
只要target值大於目前位置都有可能可以放 所以要記著  
除非找到裡面也有一樣的值那就放該值的位置  
否則放前面記錄到的可能位置  
```cpp
class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int left = 0 , right = nums.size()-1, mid;
        bool isfind = false;
        int pos = 0;
        while ( left <= right && !isfind ) {
            mid = left + ( right - left )/2;
            
            if ( target < nums[mid] ) {
                right = mid-1;
            }
            else if ( target > nums[mid] ) {
                //要記錄可能的位置 要更新
                left = mid+1;
                pos = mid+1;
            }
            else {
                isfind = true;
            }
        }

        if ( isfind ) {
            pos = mid;
        }
        return pos;

        
    }
};
```
