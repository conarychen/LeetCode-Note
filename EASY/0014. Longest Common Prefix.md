# [14. Longest Common Prefix](https://leetcode.com/problems/longest-common-prefix/description/)  
目標 -> 找出input字串的共同prefix  
每個字串的第一個比 如果都一樣就記起來 然後繼續找 直到出現第一個不一樣or有字串到尾了  
```cpp
class Solution {
public:
    string longestCommonPrefix(vector<string>& strs) {
        string result = "";
        string temp = "";
        int pos = 0;
        bool isEnd = false;
        while ( !isEnd ) {
            for ( int i = 0 ; i < strs.size() && !isEnd ; i++ ) {
                if ( temp == "" ) {
                    temp.push_back( strs[i][pos] );
                }
                else if ( pos <= strs[i].size() ) {
                    if ( pos < strs[i].size() && ( temp[0] == strs[i][pos] && i == strs.size()-1 ) ) {
                        result.push_back( temp[0] );
                        temp.pop_back();
                        pos++;
                    }
                    else if ( pos < strs[i].size() && temp[0] == strs[i][pos] ) {
                        ; //do nothing
                    }
                    else {
                        isEnd = true;
                    }
                }  
                else {
                    isEnd = true;
                }
            }
        }
        return result;
    }
};
```
