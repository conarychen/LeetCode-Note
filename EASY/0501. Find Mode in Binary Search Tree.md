# [501. Find Mode in Binary Search Tree](https://leetcode.com/problems/find-mode-in-binary-search-tree/description/)  
目標 -> 找出BST中出現最多次的數(可能有多個)  
這邊的member應該都丟private 比較安全  
反正用inorder 去找 如果有重複的就計數 因為BST的特性 如果重複出現一定連著  
如果出現新的數字了 就重製count來數  
超過紀錄就更新新的最大值 然後把該val 記下來  
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:
    void inorder( TreeNode* root) {
        if ( root == NULL ) {
            return;
        }
        
        inorder( root -> left );
        if ( pre == NULL ) {
            count = 1;
        }
        else if ( root -> val == pre -> val ) {
            count++;
        }
        else {
            count = 1;
        }

        pre = root;
        if ( count > maxCount ) {
            result.clear();
        }
        if ( count >= maxCount ) {
            maxCount = count;
            result.push_back( root -> val );
        }
        inorder( root -> right );
        return;
    }
public:
    int count = 1, maxCount=-1;
    vector<int> result;
    TreeNode* pre = NULL;
    vector<int> findMode(TreeNode* root) {
        inorder( root );
        return result;
    }
};
```
