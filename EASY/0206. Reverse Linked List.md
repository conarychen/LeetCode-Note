# [206. Reverse Linked List](https://leetcode.com/problems/reverse-linked-list/description/)  
目標 -> 將linked list反轉  
初始 pre = NULL temp -> nextN  
開始 pre -> temp -> nextN ====> pre <- temp nextN ====> () <- pre temp -> nextN  
重複這樣反轉  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode* temp = head,* nextN,* pre;
        if ( head == NULL ) {
            return NULL;
        }
        nextN = temp -> next;
        temp -> next = NULL;
        while ( nextN != NULL ) {
            pre = temp;
            temp = nextN;
            nextN = nextN -> next;
            temp -> next = pre;
        }
  
        return temp;
    }
};
```
