# [67. Add Binary](https://leetcode.com/problems/add-binary/description/)  
目標 -> 進行binary計算  
從個位數一位一位往上加  
如果加到其中一個沒了 還是要繼續 因為可能前面有進位  
```cpp
class Solution {
public:
    string addBit( char a, char b, bool &addOne ) {
        string result = "";
        if ( a == '0' && b == '0' ) {
            if ( addOne ) {
                result.append( "1" );
                addOne = false;
            }
            else {
                result.append( "0" );
            }
        }
        else if ( ( a == '0' && b == '1' ) || ( a == '1' && b == '0' ) ) {
            if ( addOne ) {
                addOne = true;
                result.append( "0" );
            }
            else {
                result.append( "1" );
            }
        }
        else if ( a == '1' && b == '1' ) {
            if ( addOne ) {
                addOne = true;
                result.append( "1" );
            }
            else {
                addOne = true;
                result.append( "0" );
            }
        }
        return result;
    }
    string addBinary(string a, string b) {
        bool addOne = false;
        string result = "";
        string temp;
        for ( int i = a.size()-1, j = b.size()-1 ; i >= -1 || j >= -1 ; ) {
            if ( i > -1 && j > -1 ) {
                result.insert( 0, addBit( a[i], b[j], addOne ) );
                i--;
                j--;
            }
            else if ( i > -1 && j == -1 ) {
                result.insert( 0, addBit( a[i], '0', addOne ) );
                i--;
            }
            else if ( i == -1 && j > -1 ) {
                result.insert( 0, addBit( '0', b[j], addOne ) );
                j--;
            }
            else if ( i == -1 && j == -1 ) {
                if ( addOne ) {
                    result.insert( 0, "1" );
                }
                i--;
                j--;
            }
        }
        return result;
    }
};
```
