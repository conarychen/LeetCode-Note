# [69. Sqrt(x)](https://leetcode.com/problems/sqrtx/description/)  
目標 -> 實作根號 不使用內建運算子  
利用binary search來找  
有可能溢位 所以long long來放  
```cpp
class Solution {
public:
    int mySqrt(int x) {
        if ( x == 1 ) {
            return 1;
        }
        long long temp = 0;
        int left = 0, right = x, mid;
        bool isDone = false;
        int result = 0;
        while( !isDone && left+1 != right ) {
            mid = left + (right-left)/2;
            temp = (long long)mid * mid;
            if ( temp == x ) {
                isDone = true;
                result = mid;
            }
            else if ( temp < x ) {
                left = mid;
                result = mid;
            }
            else {
                right = mid;
            }

        }
        return result;
    }
};
```
