# [21. Merge Two Sorted Lists](https://leetcode.com/problems/merge-two-sorted-lists/description/)  
目標 -> 合併兩條linked list (要照node val大小排)  
基本上就是merge sort的想法  
也就是用兩個指標走兩條list 比較小的就先選 選了的那條就往下走一個繼續比  
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        ListNode *head = NULL;
        ListNode *temp = new ListNode(); //dummy
        head = temp;
        while ( list1 != NULL && list2 != NULL ) {
            if ( list1 -> val <= list2 -> val ) {

                temp -> next = list1;
                list1 = list1 -> next;
            }
            else {

                temp -> next = list2;
                list2 = list2 -> next;
            }
            temp = temp -> next;
        }

        if ( list1 != NULL ) {
            temp -> next = list1;
        }
        else {
            temp -> next = list2;
        }

        temp = head;
        if ( head != NULL ) head = head -> next;
        delete temp;
        return head;
    }
};
```
