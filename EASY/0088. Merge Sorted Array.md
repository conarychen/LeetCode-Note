# [88. Merge Sorted Array](https://leetcode.com/problems/merge-sorted-array/description/)  
目標 -> 合併兩個陣列  
我這邊是單純在nums1裡找nums2元素該待的位置  
但其實用merge sort就好 也更快  
```cpp
class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        while( nums1.size() > m ) {
            nums1.pop_back();
        }

        int pos = 0;

        for ( int i = 0 ; i < nums1.size() && pos < nums2.size() ; i++ ) {
            if ( pos > nums2.size() ) {
                i = nums1.size();
            }
            else if ( nums2[pos] <= nums1[i] ) {
                
                nums1.insert( nums1.begin() + i , nums2[pos] );
                pos++;
            }
            
        }
        if ( pos < nums2.size() ) {
            
            while ( m == 0 && nums1.size() != 0 ) {
                nums1.pop_back();
            }
        
            for ( ; pos < nums2.size() ; pos++ ) {
                nums1.push_back( nums2[pos] );
            }
        }
        
    }
};
```
