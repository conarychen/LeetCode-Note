# [968. Binary Tree Cameras](https://leetcode.com/problems/binary-tree-cameras/description/)  
目標 -> 在節點上放置camera(以下用cam) cam的範圍是自己以及上下的節點  
&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;要把監視範圍覆蓋整顆二元樹 最少需要多少個cam  
用後序遍歷(postorder traversal)處理  
用node -> val 來記錄狀態 0(沒有被監視) 1(放置監視器) 2(被監視)  
所以可以知道我們要找左右child有一個以上是0的node來放監視器(1)  
當傳進來的node == NULL時 當作被監視(2)  
接著就比對情況放值 只要放了1那就count++即可  

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
private:
    int count = 0;
    // node val -> 0 -> no cam and monitor -> 1 -> cam -> 2 -> monitor
    int counting( TreeNode* root ) {
        if ( root == NULL ) {
            return 2;
        } 
        int left = counting( root -> left );
        int right = counting( root -> right );
        // 子樹只要有一個是0 代表父就必須要放cam
        if ( left == 0 || right == 0 ) {
            count++;
            return 1;
        }
        // 子樹只要有cam 就代表父不用 (除非其中一方為0 但這條件上面會執行)
        else if ( left == 1 || right == 1 ) {
            return 2;
        }
        else { //只剩下兩子都是2的情況
            return 0;
        }
    }

public:
    int minCameraCover(TreeNode* root) {
        if ( root == NULL ) {
            return 0;
        }
        
        if ( counting( root ) == 0 ) {
            count++;
        }
        return count;
    }
};
```
