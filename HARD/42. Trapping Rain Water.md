# [42. Trapping Rain Water](https://leetcode.com/problems/trapping-rain-water/description/)  
利用DP概念 從頭->尾 尾->頭 兩次走訪 紀錄每個位置的最大left和right  
最後就算體積加起來

```cpp
class Solution {
public:
    int min( int a, int b ) {
        if ( a < b ) {
            return a;
        }
        return b;
    }
    int max( int a, int b ) {
        if ( a > b ) {
            return a;
        }
        return b;
    }
    int trap(vector<int>& height) {
        int result = 0, temp;
        vector<int> left(height.size()), right(height.size());
        left[0] = height[0];
        right[right.size()-1] = height[height.size()-1];
        // 左到右用vector紀錄每個位置的最大left
        for ( int i = 1 ; i < height.size() ; i++ ) {
            left[i] = max( left[i-1], height[i] );

        }
        // 同上 只是改成右到左找最大left
        for ( int i = height.size()-2 ; i >= 0 ; i-- ) {
            right[i] = max( right[i+1], height[i] );

        }
        //最後算出體積加起來就好
        for ( int i = 0 ; i < height.size() ; i++ ) {
            temp = ( min( left[i], right[i] ) - height[i] );
            if ( temp > 0 ) {
                result += temp;
            }
        }
        return result;   

    }
};
```
