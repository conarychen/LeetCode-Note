# [239. Sliding Window Maximum](https://leetcode.com/problems/sliding-window-maximum/description/)  
目標 -> 在大小為k的視窗內取最大值 取完整個視窗往右滑動一格 直到視窗右端超過nums 並記錄每次取到的局部最大  
詳細在下面註解中  
概略想法是 利用deque 留下局部最大  
只要視窗範圍離開目前最大值 就把deque的top pop掉 換新的局部最大  
每次滑動都有新的值要來維護是否為局部最大  
放進deque時從後往前比 如果遇到比較小的就pop掉  
直到遇到比較大的就push進去

```cpp
class Solution {
private:
    deque<int> dq;
    // ex[1,3,-1,-3,5,3,6,7] k=3
    // 保持進入的順序 且只留局部最大
    // 以第一次加入3個為例 括起來的數字代表被刪掉
    // dq ==> 1 -> (1)3 -> 3,-1 (這邊以值來看比較直接 但存是存index)
    // 1進入時deque為空 直接放
    // 3進入時 由deque尾往頭開始比 比他小的都可以丟掉 直到比他大的 所以1被丟了
    // -1進入時 因為尾巴開始比沒有比他小的所以就直接放入
    // dq等於是只留下會成為最大值的候補 如果後面有範圍內且值更大的就代表此時的最大
    //值不可能是前者 那就可以丟棄前者保留後者

    void dqpush( int num, vector<int>& nums ) {
        while ( dq.size() != 0 && nums[num] > nums[dq.back()] ) {
            
            dq.pop_back();
        }
        dq.push_back( num );
    }
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        if ( k == 1 ) {
            return nums;
        }
        vector<int> result;
        for ( int i = 0 ; i < k ; i++ ) {
            dqpush( i, nums );
        }
        result.push_back( nums[dq.front()] );
        for ( int i = k, point = 1 ; i < nums.size() ; i++, point++ ) {
            while ( dq.front() < point ) {
                dq.pop_front();
            }
            dqpush( i, nums );
            result.push_back( nums[dq.front()] );
        }
        return result;
    }
};
```
