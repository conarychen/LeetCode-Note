# [4. Median of Two Sorted Arrays](https://leetcode.com/problems/median-of-two-sorted-arrays/submissions/2030623199/)
這題主要就是將兩個vector合併後，找到中位數即可  
可以直接使用STL的merge sort

```cpp
class Solution {
public:

    vector<int> mergeSort( vector<int>& nums1, vector<int>& nums2 ) {
        int m = 0, n = 0;
        vector<int> result;
        while( m < nums1.size() && n < nums2.size() ) { // Sort two vector
            if ( nums1[m] <= nums2[n] ) {
                result.push_back( nums1[m] );
                m++;
            }
            else {
                result.push_back( nums2[n] );
                n++;
            }
        }

        
        if ( m < nums1.size() ) {  //put all left nums in vector
            for ( ; m < nums1.size() ; m++ ) {
                result.push_back( nums1[m] );
            }
        }
        else if ( n < nums2.size() ) {
            for ( ; n < nums2.size() ; n++ ) {
                result.push_back( nums2[n] );
            }
        }
        return result;
    }

    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        double result = 0;
        vector<int> temp;
        temp = mergeSort( nums1, nums2 );
        
        if ( temp.size() % 2 == 1 ) { // Odd
            result = temp[(temp.size()-1)/2.0];
        }
        else { //Even
            result = ( temp[(temp.size())/2.0] + temp[((temp.size())/2.0)-1] ) / 2.0;
        }
        return result;
    }

};

```
